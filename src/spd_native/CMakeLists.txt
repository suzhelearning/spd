cmake_minimum_required(VERSION 3.20)
project(spd_native LANGUAGES CXX)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tianji_spd_interfaces REQUIRED)
find_package(Python REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11 CONFIG REQUIRED)
find_path(MUJOCO_INCLUDE_DIR mujoco/mujoco.h REQUIRED)
find_library(MUJOCO_LIBRARY mujoco REQUIRED)

pybind11_add_module(_spd_native src/module.cpp src/physics.cpp src/executor.cpp src/control.cpp)
target_compile_features(_spd_native PRIVATE cxx_std_17)
target_include_directories(_spd_native PRIVATE include "${MUJOCO_INCLUDE_DIR}")
target_link_libraries(_spd_native PRIVATE "${MUJOCO_LIBRARY}")
ament_target_dependencies(_spd_native PUBLIC rclcpp std_msgs tianji_spd_interfaces)
set_target_properties(_spd_native PROPERTIES INSTALL_RPATH "$ENV{CONDA_PREFIX}/lib;${CMAKE_INSTALL_PREFIX}/lib")

add_executable(spd_executor src/main.cpp)
target_compile_features(spd_executor PRIVATE cxx_std_17)
target_link_libraries(spd_executor PRIVATE pybind11::embed)
set_target_properties(spd_executor PROPERTIES INSTALL_RPATH "$ENV{CONDA_PREFIX}/lib")
install(TARGETS _spd_native LIBRARY DESTINATION "lib/python${Python_VERSION_MAJOR}.${Python_VERSION_MINOR}/site-packages")
install(TARGETS spd_executor RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_environment_hooks("${CMAKE_CURRENT_SOURCE_DIR}/pythonpath.dsv.in")
ament_package()
